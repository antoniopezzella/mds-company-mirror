import { useState } from 'react';
import { Rating, Stack } from '@mui/material';

# Rating

export function Demo() {
  const [value, setValue] = useState(3);
  return (
    <Stack direction="row" spacing={2}>
      <Rating value={value} onChange={(_, v) => setValue(v ?? 0)} />
    </Stack>
  );
}
