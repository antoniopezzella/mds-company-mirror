import React from 'react';
export function Example() {
  const [value, setValue] = React.useState(0);
  return (
    <Tabs value={value} onChange={(_, v) => setValue(v)}>
      <Tab label="Tab One" />
      <Tab label="Tab Two" />
      <Tab label="Tab Three" />
    </Tabs>
  );
}
